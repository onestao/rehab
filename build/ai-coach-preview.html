折叠交互 — 实现伪代码

renderHeroInsight(ctx) {
  // 始终渲染
  renderHeader(ctx);         // 标题 + badge + chevron
  renderBaseline(ctx);       // 指标条 + 共享进度组件 compact

  // 点击 header 切换展开/收起
  header.onclick = () => {
    if (expanded) {
      collapse();            // max-height → 0, opacity → 0
    } else {
      expand();
      if (!bodyRendered) {   // 首次展开时才渲染
        renderAnalysis(ctx);
        renderLLMSkeleton();
        streamAdvice(ctx).then(swap).catch(fallback);
        bodyRendered = true;
      }
    }
  };
  // CTA 按钮阻止冒泡,不触发折叠
  actions.onclick = (e) => e.stopPropagation();
}

// baseline 不依赖 LLM,立刻可见
renderBaseline({ plan, metrics }) {
  renderMetricStrip(metrics);            // 3 个 glass tile
  renderSharedPlanProgress(plan,'compact'); // 1 行 + 微进度条
}
